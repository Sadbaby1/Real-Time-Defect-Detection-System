#YOLO

Models: train, val, predict
        --> train : training the model
        --> val : evaluate and see the performance of the model
        --> predict : for performing detection on unsine dataset

Tasks : detect, segment, classify
        --> detect : object detection task
        --> segment : for image gegmentation
        --> classify : for classiftication

data: format can difer for task type: data.yaml data_folder, dataset_name 

------------------------------------------------------------------>>>>>
commands cmd:
        --> detect : !yolo task=detect mode=predict model=yolov8n.pt source="image_path"
        --> segment : !yolo task=segement mode=predict model=yolov8n-seg.pt source="image_path"
        --> classify : !yolo task=classify mode=predict model=yolov8n-cls.pt source="image_path"

    if result.boxes is not None:
        for box, cls_id, conf in zip(result.boxes.xyxy, result.boxes.cls, result.boxes.conf):
            x1, y1, x2, y2 = map(int, box)
            label = f"{model.names[int(cls_id)]} {conf:.2f}"
            cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)
            cv2.putText(frame, label, (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

    # 2. Segmentation mask contours
    if result.masks is not None:
        for segment in result.masks.xy:
            if len(segment) > 0:
                pts = segment.astype(np.int32)
                cv2.polylines(frame, [pts], isClosed=True, color=(255, 0, 0), thickness=2)
